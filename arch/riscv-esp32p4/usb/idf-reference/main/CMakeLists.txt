set(reference "$ENV{IDF_PATH}/examples/peripherals/usb/device/tusb_serial_device/main")
if(NOT EXISTS "${reference}/tusb_serial_device_main.c")
    message(FATAL_ERROR "The reviewed ESP-IDF serial-device example is missing")
endif()
file(SHA256 "${reference}/tusb_serial_device_main.c" reference_sha)
if(NOT reference_sha STREQUAL "91c7d0f862bb86b01f821340cb075334ea820813148161e431d67d0486c72a58")
    message(FATAL_ERROR "The official serial-device example differs from reviewed IDF6.0.1")
endif()
option(HS_REFERENCE_PERSIST "Single-boot staged diagnostic in backed-up app sector" OFF)
option(HS_REFERENCE_INJECT_INSTALL_ERROR "Headless trace self-test, not a USB fault" OFF)
option(HS_REFERENCE_LINK_TRACE "Read-only PHY snapshots and queued USB event counters" OFF)
option(HS_REFERENCE_FORCE_FS_HSPHY "Diagnostic: full speed on the native HS UTMI PHY" OFF)
option(HS_REFERENCE_RECONNECT "Diagnostic: one delayed soft detach/attach" OFF)
if(HS_REFERENCE_RECONNECT AND
   (NOT HS_REFERENCE_PERSIST OR NOT HS_REFERENCE_LINK_TRACE OR
    HS_REFERENCE_INJECT_INSTALL_ERROR OR HS_REFERENCE_FORCE_FS_HSPHY))
    message(FATAL_ERROR "Reconnect comparison requires persistent HS link trace without injection or forced FS")
endif()
if(HS_REFERENCE_FORCE_FS_HSPHY AND
   (NOT HS_REFERENCE_PERSIST OR NOT HS_REFERENCE_LINK_TRACE OR HS_REFERENCE_INJECT_INSTALL_ERROR))
    message(FATAL_ERROR "FS-on-HS comparison requires persistent link trace and no error injection")
endif()
if(HS_REFERENCE_INJECT_INSTALL_ERROR AND NOT HS_REFERENCE_PERSIST)
    message(FATAL_ERROR "Error injection requires the explicit staged trace")
endif()
set(reference_sources "observer.c")
if(HS_REFERENCE_LINK_TRACE)
    list(APPEND reference_sources "link_events.c")
endif()
if(HS_REFERENCE_PERSIST)
    string(LENGTH "${HS_REFERENCE_PREIMAGE_CRC32}" crc_length)
    if(NOT crc_length EQUAL 10 OR NOT HS_REFERENCE_PREIMAGE_CRC32 MATCHES "^0x[0-9A-Fa-f]+$")
        message(FATAL_ERROR "Persistent reference needs CRC32 of retained app scratch preimage")
    endif()
    list(APPEND reference_sources "journal.c" "trace.c" "journal_idf.c" "flash_window.c")
endif()
idf_component_register(SRCS ${reference_sources} INCLUDE_DIRS "." "${reference}"
                       REQUIRES esp_partition esp_hw_support esp_driver_usb_serial_jtag bootloader_support esp_timer)
if(HS_REFERENCE_LINK_TRACE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE HS_REFERENCE_LINK_TRACE=1)
endif()
if(HS_REFERENCE_FORCE_FS_HSPHY)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE HS_REFERENCE_FORCE_FS_HSPHY=1)
endif()
if(HS_REFERENCE_RECONNECT)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE HS_REFERENCE_RECONNECT=1)
endif()
if(HS_REFERENCE_PERSIST)
    target_link_options(${COMPONENT_LIB} INTERFACE
        "LINKER:--wrap=esp_partition_main_flash_region_safe"
        "LINKER:-u,__wrap_esp_partition_main_flash_region_safe")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE HS_REFERENCE_PERSIST=1
                               HS_REFERENCE_PREIMAGE_CRC32=${HS_REFERENCE_PREIMAGE_CRC32}u)
    # Explicit IDF size profile, not altered flash bounds or weakened checks.
    if(NOT CONFIG_COMPILER_OPTIMIZATION_SIZE)
        message(FATAL_ERROR "Staged trace needs sdkconfig.trace size profile; scratch boundary stays fixed")
    endif()
    if(HS_REFERENCE_INJECT_INSTALL_ERROR)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE HS_REFERENCE_INJECT_INSTALL_ERROR=1)
    endif()
endif()
